rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /users/{uid} {
      allow read: if request.auth != null
        && (request.auth.uid == uid
            || request.auth.token.email == 'lanterns@yandex.ru');
      allow write: if request.auth != null
        && request.auth.uid == uid;

      match /{sub=**} {
        allow read, write: if request.auth != null
          && request.auth.uid == uid;
      }
    }

    match /feedback/{id} {
      allow create: if request.auth != null
        && request.resource.data.role in ['user','admin']
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 4000
        && ((request.resource.data.role == 'user' && request.resource.data.threadId == request.auth.uid)
            || (request.resource.data.role == 'admin' && request.auth.token.email == 'lanterns@yandex.ru'));

      allow read: if request.auth != null
        && (resource.data.threadId == request.auth.uid
            || request.auth.token.email == 'lanterns@yandex.ru');

      allow update: if request.auth != null
        && (request.auth.token.email == 'lanterns@yandex.ru'
            || (resource.data.threadId == request.auth.uid
                && resource.data.role == 'admin'
                && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['readByUser'])));

      allow delete: if request.auth != null
        && request.auth.token.email == 'lanterns@yandex.ru';
    }
  }
}
